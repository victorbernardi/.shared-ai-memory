def check_recency_report():
    """LÃª o relatÃ³rio recency_status.md do shared e alerta se houver fontes desatualizadas.
    Retorna (outdated, all_sources) onde all_sources Ã© um dicionÃ¡rio {fonte: status_modificacao}."""
    recency_file = _shared_dir / "recency_status.md"
    outdated = []
    all_sources = {}
    
    if not recency_file.exists():
        print(f"INFO: RelatÃ³rio de recÃªncia nÃ£o encontrado em {recency_file}. Pulando validaÃ§Ã£o.")
        return outdated, all_sources
        
    try:
        with recency_file.open("r", encoding="utf-8") as f:
            lines = f.readlines()
            
        # Parse da tabela Markdown
        table_started = False
        table_lines = []
        for line in lines:
            stripped = line.strip()
            if stripped.startswith("|"):
                table_lines.append(stripped)
                if "Status de Rec" in stripped:
                    table_started = True
                    continue
                if table_started and not any(x in stripped for x in [":---", "---:"]):
                    # Ã‰ uma linha de dados
                    parts = [p.strip() for p in stripped.split("|")]
                    # parts[0] Ã© vazio, parts[1]=Fonte, parts[2]=Arquivo, parts[3]=Status, parts[4]=Ãšltima ModificaÃ§Ã£o
                    if len(parts) >= 5:
                        fonte = parts[1].strip()
                        arquivo = parts[2].strip()
                        status = parts[3].strip()
                        ultima_mod = parts[4].strip()
                        all_sources[fonte] = {
                            "arquivo": arquivo,
                            "status": status,
                            "ultima_modificacao": ultima_mod
                        }
                        if "ðŸŸ¡ Desatualizado" in status or "ðŸ”´ Ausente" in status:
                            outdated.append(fonte)
                            
        if outdated:
            print("\n" + "=" * 80)
            print("âš ï¸  ALERTA DE GOVERNANÃ‡A: HÃ FONTES DESATUALIZADAS OU AUSENTES!")
            print("=" * 80)
            print("As seguintes fontes analÃ­ticas estÃ£o obsoletas e requerem atenÃ§Ã£o:")
            for f in outdated:
                print(f"  - ðŸ”´ {f}")
            print("\nConsulte a tabela de recÃªncia abaixo para detalhes:")
            print("-" * 80)
            for t_line in table_lines:
                print(t_line)
            print("=" * 80)
            print("AVISO: Prosseguindo com a consolidaÃ§Ã£o da BUP pÃ³s-venda...\n")
        else:
            print("\nðŸŸ¢ SUCESSO: Todas as fontes de dados de ingestÃ£o estÃ£o atualizadas!\n")
            
    except Exception as e:
        print(f"AVISO: Erro ao analisar o relatÃ³rio de recÃªncia: {e}")
        
    return outdated, all_sources


def generate_scheduler_log_content(onedrive_status, recency_data, metricas, timestamp=None):
    """Gera o bloco formatado do log de agendamento em modo cumulativo."""
    if timestamp is None:
        timestamp = datetime.now().strftime("%Y-%m-%d %H:%M")
        
    status_onedrive_str = "SUCESSO"
    if "FALHA" in onedrive_status:
        status_onedrive_str = f"FALHA ({onedrive_status})"
    elif onedrive_status != "OK":
        status_onedrive_str = f"AVISO ({onedrive_status})"
        
    linhas_fontes = []
    # Mapeia as chaves do recency_data para a exibiÃ§Ã£o no log
    for fonte, info in recency_data.items():
        arq = info.get("arquivo", "-")
        mod = info.get("ultima_modificacao", "-")
        stat = info.get("status", "ok")
        # Simplifica o status para exibiÃ§Ã£o
        stat_lbl = "OK"
        if "Desatualizado" in stat or "obsoleto" in stat.lower():
            stat_lbl = "DESATUALIZADO"
        elif "Ausente" in stat or "ausente" in stat.lower():
            stat_lbl = "AUSENTE"
            
        linhas_fontes.append(f"- {fonte}: {arq} | ModificaÃ§Ã£o: {mod} ({stat_lbl})")
        
    linhas_fontes_str = "\n".join(linhas_fontes)
    
    clientes_total = metricas.get("clientes_total", 0)
    clientes_conv = metricas.get("clientes_convertidos", 0)
    vendedores = metricas.get("consultores_ativos", 0)
    
    log_block = f"""================================================================================
RODADA DE ATUALIZAÃ‡ÃƒO BUP: {timestamp}
================================================================================
[STATUS DA TAREFA]
- Entrega no OneDrive: {status_onedrive_str}
- MÃ©tricas: {clientes_total} clientes totais | {clientes_conv} oportunidades | {vendedores} consultores ativos

[RECÃŠNCIA E ORIGEM DAS TABELAS]
{linhas_fontes_str}
--------------------------------------------------------------------------------
"""
    return log_block

def _formatar_e_validar_excel(path: _Path) -> None:
    wb = load_workbook(path)
    ws = wb.active
    max_row = ws.max_row

    header = {ws.cell(row=1, column=c).value: c for c in range(1, ws.max_column + 1)}
    
    colunas_filipe = [
        "Data_Tentativa_1", "Status_Contato_1",
        "Data_Tentativa_2", "Status_Contato_2", "Observacao"
    ]
    colunas_editaveis = set(colunas_filipe)

    # Estilos de cores
    fill_black = PatternFill(start_color="000000", end_color="000000", fill_type="solid")
    font_yellow = Font(name="Calibri", size=11, bold=True, color="FFFFC000") # Amarelo tom cevap/leads-csc
    
    fill_orange = PatternFill(start_color="FFFFC000", end_color="FFFFC000", fill_type="solid")
    border_dotted = Border(
        left=Side(style="dotted", color="808080"),
        right=Side(style="dotted", color="808080"),
        top=Side(style="dotted", color="808080"),
        bottom=Side(style="dotted", color="808080")
    )

    # Formatar CabeÃ§alho (Preto com letras Amarelas)
    for col_idx in range(1, ws.max_column + 1):
        cell = ws.cell(row=1, column=col_idx)
        cell.fill = fill_black
        cell.font = font_yellow

    # Aplicar AutoFilter
    ws.auto_filter.ref = ws.dimensions

    # Auto-width
    for col in ws.columns:
        col_letter = get_column_letter(col[0].column)
        col_name = ws.cell(row=1, column=col[0].column).value
        if col_name == "Equipamentos":
            ws.column_dimensions[col_letter].width = 45
        else:
            max_len = max((len(str(cell.value or "")) for cell in col), default=0)
            ws.column_dimensions[col_letter].width = min(max(max_len + 3, 12), 60)

    # Formatar colunas editÃ¡veis
    for col_name in colunas_editaveis:
        if col_name not in header:
            continue
        col_idx = header[col_name]
        for row in ws.iter_rows(min_row=2, max_row=max_row, min_col=col_idx, max_col=col_idx):
            row[0].fill = fill_orange
            row[0].border = border_dotted

    # Formatar SOW
    if "SOW" in header:
        sow_col = header["SOW"]
        for row in ws.iter_rows(min_row=2, max_row=max_row, min_col=sow_col, max_col=sow_col):
            row[0].number_format = "0.0%"

    # ValidaÃ§Ã£o Dropdowns Estrita para Status
    opcoes = '"Pendente,Venda,NÃ£o Venda"'
    for col_name in ["Status_Contato_1", "Status_Contato_2"]:
        if col_name not in header:
            continue
        col_letter = ws.cell(row=1, column=header[col_name]).column_letter
        dv = DataValidation(
            type="list",
            formula1=opcoes,
            allow_blank=True,
            showErrorMessage=True,
            errorTitle="Valor invÃ¡lido",
            error="Selecione um status vÃ¡lido da lista: Pendente, Venda ou NÃ£o Venda.",
            errorStyle="stop"
        )
        dv.add(f"{col_letter}2:{col_letter}{max_row}")
        ws.add_data_validation(dv)

    # ValidaÃ§Ã£o de Data (+-30 dias)
    hoje = date.today()
    min_date = hoje - timedelta(days=30)
    max_date = hoje + timedelta(days=30)
    serial_min = (min_date - date(1899, 12, 30)).days
    serial_max = (max_date - date(1899, 12, 30)).days
    msg_erro = (
        f"Informe uma data entre {min_date.strftime('%d/%m/%Y')} "
        f"e {max_date.strftime('%d/%m/%Y')}."
    )
    for col_name in ["Data_Tentativa_1", "Data_Tentativa_2"]:
        if col_name not in header:
            continue
        col_idx = header[col_name]
        col_letter = ws.cell(row=1, column=col_idx).column_letter
        for row in ws.iter_rows(min_row=2, max_row=max_row, min_col=col_idx, max_col=col_idx):
            row[0].number_format = "DD/MM/YYYY"
        
        dv_data = DataValidation(
            type="date",
            operator="between",
            formula1=str(serial_min),
            formula2=str(serial_max),
            allow_blank=True,
            showErrorMessage=True,
            errorTitle="Data invÃ¡lida",
            error=msg_erro,
            errorStyle="stop"
        )
        dv_data.add(f"{col_letter}2:{col_letter}{max_row}")
        ws.add_data_validation(dv_data)

    # Garantir que a planilha nÃ£o esteja protegida por senha para liberar ordenaÃ§Ã£o na Web
    ws.protection.sheet = False
    ws.protection.disable()

    wb.save(path)

                return f"{mun_clean} / {est_clean}"
