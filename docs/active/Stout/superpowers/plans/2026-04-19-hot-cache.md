**Tech Stack:** Python 3.11+, subprocess, gemini CLI (já disponível em PATH), pytest

  ```python
  def generate_hot_cache(summary: "SessionSummary", wiki_dir: Path) -> bool:
      """
      Gera wiki_dir/hot.md com resumo da sessão via Gemini headless.
      Retorna True se bem-sucedido, False em qualquer falha (best-effort).
      """
      if shutil.which("gemini") is None:
          return False

      topics_txt = "\n".join(f"- {t}" for t in summary.topics) or "(nenhum)"
      tasks_done_txt = "\n".join(f"- {t}" for t in summary.tasks_completed) or "(nenhuma)"
      files_txt = "\n".join(f"- {f['path']} [{f.get('action', '?')}]" for f in summary.files_modified) or "(nenhum)"
      errors_txt = "\n".join(
          f"- Erro: {e['error']} → Solução: {e.get('solution', '?')}"
          for e in summary.errors_resolved
      ) or ""
      pending_txt = "\n".join(
          f"- {t.description if hasattr(t, 'description') else str(t)}"
          for t in summary.tasks_pending
      ) or "(nenhuma)"

      no_errors_note = "Não há erros registrados; omita a seção 'Erros a evitar'." if not summary.errors_resolved else ""

      prompt = f"""Você é um assistente que gera um arquivo hot.md (hot cache) para retomada de sessão de trabalho com IA.
  Com base nos dados da sessão abaixo, gere um resumo em português em até 500 palavras com estas seções:

  ## O que foi feito
  ## Próximos passos
  ## Erros a evitar

  {no_errors_note}
  Responda SOMENTE com o conteúdo markdown (sem bloco de código, sem explicações adicionais).
  Comece com: # Hot Cache — {summary.date}

  ---
  Tópicos:
  {topics_txt}

  Tarefas concluídas:
  {tasks_done_txt}

  Arquivos modificados:
  {files_txt}

  Erros resolvidos:
  {errors_txt if errors_txt else "(nenhum)"}

  Próximas tarefas:
  {pending_txt}
  """

      try:
          result = subprocess.run(
              ["gemini", "-p", prompt],
              capture_output=True,
              text=True,
              encoding="utf-8",
              timeout=30,
          )
          if result.returncode != 0:
              return False

          content = result.stdout.strip()
          if not content:
              return False

          wiki_dir.mkdir(parents=True, exist_ok=True)
          (wiki_dir / "hot.md").write_text(content, encoding="utf-8")
          return True

      except (subprocess.TimeoutExpired, FileNotFoundError, OSError):
          return False
  ```
