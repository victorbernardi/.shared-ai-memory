```python
import pandas as pd
from datetime import date
from pathlib import Path
import sys

sys.path.append(str(Path(__file__).parents[2]))
from src.config import SHARED_DATA

_XLSX = SHARED_DATA / "tabela_orçamentos_cancelados.xlsx"
_COLS = ["Codigo_Peca", "Num_Orc", "Cliente", "Filial", "Data_Orc", "Valor_Cancelado", "Motivo"]


def _carregar_xlsx() -> pd.DataFrame:
    df = pd.read_excel(_XLSX)
    df.columns = _COLS
    df["Valor_Cancelado"] = pd.to_numeric(df["Valor_Cancelado"], errors="coerce").fillna(0)
    df["Data_Orc"] = pd.to_datetime(df["Data_Orc"], errors="coerce").dt.date
    return df


def cancelamentos_ontem(
    data_ref: date | None = None,
    threshold: float = 50_000.0,
) -> dict | None:
    if data_ref is None:
        from datetime import datetime, timedelta
        data_ref = (datetime.now() - timedelta(days=1)).date()

    df = _carregar_xlsx()
    df_ontem = df[df["Data_Orc"] == data_ref].copy()

    total = df_ontem["Valor_Cancelado"].sum()
    if total < threshold:
        return None

    top3 = (
        df_ontem.groupby("Cliente")["Valor_Cancelado"]
        .sum()
        .sort_values(ascending=False)
        .head(3)
        .to_dict()
    )

    motivo_counts = df_ontem["Motivo"].value_counts()
    motivo_dominante = motivo_counts.index[0] if not motivo_counts.empty else "N/A"
    pct_motivo = motivo_counts.iloc[0] / len(df_ontem) * 100 if not df_ontem.empty else 0.0

    return {
        "data": data_ref,
        "total": total,
        "top3_clientes": top3,
        "motivo_dominante": motivo_dominante,
        "pct_motivo": pct_motivo,
    }


if __name__ == "__main__":
    resultado = cancelamentos_ontem()
    if resultado:
        print(f"Cancelamentos de {resultado['data']}: R$ {resultado['total']:,.0f}")
    else:
        print("Sem cancelamentos relevantes.")
```

```python
def _cancelamentos_md(dados: dict | None) -> str:
    if dados is None:
        return ""
    top3_linhas = ", ".join(
        f"{nome} (R$ {_brl(valor)})"
        for nome, valor in dados["top3_clientes"].items()
    )
    return (
        f"\n---\n\n"
        f"## ⚠️ Cancelamentos de ontem — {dados['data'].strftime('%d/%m/%Y')}\n\n"
        f"**Total cancelado:** R$ {_brl(dados['total'])}\n"
        f"**Top clientes:** {top3_linhas}\n"
        f"**Motivo dominante:** {dados['motivo_dominante']} ({dados['pct_motivo']:.1f}%)\n\n"
    )
```
